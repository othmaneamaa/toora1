interface Station {
  title: string;
  coordinate: { latitude: number; longitude: number };
  availableBikes: [{ title: string; battery: string; price: string }];
}

export default function Home({ navigation }: any) {
  const [modalVisible, setModalVisible] = useState(false);
  const [selectedStation, setSelectedStation] = useState<Station>({
    title: "",
    coordinate: { latitude: 0, longitude: 0 },
    availableBikes: [{ title: "", battery: "", price: "" }],
  });
  const [stations, setStations] = useState<Station[]>([]);
  const [loading, setLoading] = useState(true);

  const paymentContext = useContext(PaymentContext);

  useEffect(() => {
    loadStations();
  }, []);

  const loadStations = async () => {
    setLoading(true);
    
    try {
      console.log("[Home] Chargement stations...");
      const apiStations = await getStations();
      
      if (apiStations && apiStations.length > 0) {
        setStations(apiStations);
        console.log(`[Home] ${apiStations.length} stations chargées`);
      } else {
        console.log("[Home] API vide, données locales");
        setStations(backupStations);
      }
    } catch (error) {
      console.error("[Home] Erreur:", error);
      setStations(backupStations);
    } finally {
      setLoading(false);
    }
  };

  if (loading) {
    return (
      <View style={[styles.container, styles.loadingContainer]}>
        <ActivityIndicator size="large" color={Colors.primary} />
        <Text style={styles.loadingText}>Chargement des stations...</Text>
      </View>
    );
  }

  return (
    <View style={styles.container}>
      <MapView
        initialRegion={{
          latitude: 33.573110,
          longitude: -7.589843,
          latitudeDelta: 0.1,
          longitudeDelta: 0.1,
        }}
        style={styles.map}
        showsBuildings={false}
        mapType="terrain"
      >
        {stations.map((station, index) => (
          <Marker
            key={index}
            coordinate={station.coordinate}
            title={station.title}
            description={`${station.availableBikes.length} vélos disponibles`}
            onPress={() => {
              setSelectedStation(station);
              setModalVisible(true);
            }}
          >
            <View style={styles.marker}>
              <FontAwesome5 name="bicycle" size={20} color={Colors.white} />
              <Text style={styles.markerCount}>
                {station.availableBikes.length}
              </Text>
            </View>
          </Marker>
        ))}
      </MapView>
      
      <RentModal
        navigation={navigation}
        visible={modalVisible}
        selectedStation={selectedStation}
        onPress={() => setModalVisible(!modalVisible)}
      />
      
      {paymentContext &&
        paymentContext.paymentData &&
        paymentContext.paymentData.paymentSuccess &&
        !paymentContext.paymentData.connectionSuccess && (
          <ConnectionModal
            bikeTitle={paymentContext.paymentData.bikeTitle}
            bikeBattery={paymentContext.paymentData.bikeBattery}
            onConnectionOK={() =>
              paymentContext.setPaymentData?.((prev: any) => ({
                ...prev,
                connectionSuccess: true,
              }))
            }
          />
        )}
    </View>
  );
}

/**
 * Charge les stations AVEC LEURS VÉLOS RÉELS depuis MySQL
 * Format attendu par Home.tsx:
 * {
 *   title: "ST-WS100",
 *   coordinate: { latitude: 33.5731, longitude: -7.5898 },
 *   availableBikes: [{ title: "V100-01", battery: "100", price: "6.00" }]
 * }
 */
export const getStations = async () => {
  try {
    const currentUrl = await getDynamicApiUrl();
    console.log(`[API] 📡 Chargement depuis: ${currentUrl}`);
    
    const api = axios.create({ 
      baseURL: currentUrl,
      timeout: 15000 
    });
    
    // 1. Récupérer les stations
    const stationsResponse = await api.get("/api/stations");
    const mysqlStations = stationsResponse.data || [];
    
    console.log(`[API] 📊 ${mysqlStations.length} stations reçues`);
    
    if (mysqlStations.length === 0) {
      console.log("[API] ⚠️ MySQL vide, données locales");
      return localStations;
    }
    
    console.log("[API] 🔍 Première station brute:", mysqlStations[0]);
    
    // 2. Pour chaque station, récupérer ses vélos RÉELS
    const formattedStations = await Promise.all(
      mysqlStations.map(async (station) => {
        try {
          // CORRECTION : "name" → "title"
          const stationTitle = station.name || `Station ${station.id}`;
          
          // 3. Récupérer les vélos RÉELS de cette station
          let realBikes = [];
          try {
            const bikesResponse = await api.get(`/api/bikes/${station.id}`);
            realBikes = bikesResponse.data || [];
            console.log(`[API] 🚲 Station ${station.id}: ${realBikes.length} vélos réels`);
          } catch (bikeError) {
            console.warn(`[API] ⚠️ Pas de vélos pour station ${station.id}:`, bikeError.message);
            realBikes = [];
          }
          
          // 4. Formater les vélos pour Home.tsx
          const availableBikes = realBikes
            .filter(bike => bike.available === true || bike.available === 1)
            .map(bike => ({
              title: bike.name || `Vélo ${bike.id}`, // "V100-01"
              battery: bike.battery ? bike.battery.toString() : "100",
              price: bike.price ? bike.price.toString() : "6.00"
            }));
          
          // 5. Si pas de vélos réels, créer des vélos virtuels basés sur capacité
          let finalBikes = availableBikes;
          if (availableBikes.length === 0) {
            console.log(`[API] ⚠️ Aucun vélo réel, création virtuelle pour ${stationTitle}`);
            const bikeCount = Math.min(station.capacity || 3, 5);
            for (let i = 1; i <= bikeCount; i++) {
              finalBikes.push({
                title: `V${station.id}-${i.toString().padStart(2, '0')}`,
                battery: Math.floor(Math.random() * 50 + 50).toString(), // 50-100%
                price: "6.00"
              });
            }
          }
          
          // 6. Retourner la station formatée pour Home.tsx
          return {
            title: stationTitle,
            coordinate: {
              latitude: parseFloat(station.latitude) || 33.5731,
              longitude: parseFloat(station.longitude) || -7.5898
            },
            availableBikes: finalBikes
          };
          
        } catch (stationError) {
          console.error(`[API] ❌ Erreur station ${station.id}:`, stationError.message);
          
          // Fallback pour cette station
          return {
            title: station.name || `Station ${station.id}`,
            coordinate: {
              latitude: 33.5731,
              longitude: -7.5898
            },
            availableBikes: [{ title: "Vélo 1", battery: "100", price: "6.00" }]
          };
        }
      })
    );
    
    // 7. Filtrer les stations valides
    const validStations = formattedStations.filter(station => 
      station && station.title && station.coordinate
    );
    
    console.log(`[API] ✅ ${validStations.length} stations formatées`);
    console.log("[API] 📍 Exemple formaté:", validStations[0]);
    
    return validStations.length > 0 ? validStations : localStations;
    
  } catch (error) {
    console.error("[API] ❌ Erreur:", error.message);
    console.log("[API] 🏠 Données locales");
    return localStations;
  }
};

function getNetworkIP() {
  console.log("🔍 Recherche de l'adresse IP réseau...");
  
  const interfaces = os.networkInterfaces();
  let wifiIP = null;
  let ethernetIP = null;
  let otherIP = null;

  for (const interfaceName in interfaces) {
    for (const iface of interfaces[interfaceName]) {
      if (iface.family === 'IPv4' && !iface.internal) {
        console.log(`   📡 ${interfaceName}: ${iface.address}`);
        
        // Priorité: Wi-Fi
        if (interfaceName.includes('Wi-Fi') || interfaceName.includes('WiFi') || interfaceName.includes('WLAN')) {
          wifiIP = iface.address;
        }
        // Ensuite: Ethernet
        else if (interfaceName.includes('Ethernet') && iface.address.startsWith('20.')) {
          ethernetIP = iface.address;
        }
        // Autre IP
        else if (!iface.address.startsWith('169.254')) { // Pas d'APIPA
          otherIP = iface.address;
        }
      }
    }
  }

  // Retourne par ordre de priorité
  if (wifiIP) {
    console.log(`✅ IP WiFi sélectionnée: ${wifiIP}`);
    return wifiIP;
  }
  if (ethernetIP) {
    console.log(`✅ IP Ethernet sélectionnée: ${ethernetIP}`);
    return ethernetIP;
  }
  if (otherIP) {
    console.log(`✅ Autre IP sélectionnée: ${otherIP}`);
    return otherIP;
  }

  console.log("📍 Utilisation de localhost");
  return 'localhost';
}

          console.error("❌ Erreur login:", err);

        const user = results[0];

          return res.status(401).json({ 
            success: false, 
            message: "Email ou mot de passe incorrect" 
          });

        const token = jwt.sign(
          { userId: user.id, email: user.email, name: user.name },
          JWT_SECRET,
          { expiresIn: "7d" }
        );

        res.json({
          success: true,
          message: "Connexion réussie",
          token,
          user: {
            id: user.id,
            email: user.email,
            name: user.name,
            phone: user.phone || null
          }
        });
      }
    );
  } catch (error) {

app.get("/api/profile", (req, res) => {
  const authHeader = req.headers["authorization"];
  const token = authHeader && authHeader.split(" ")[1];

  if (!token) {
    return res.status(401).json({ error: "Token manquant" });
  }

  jwt.verify(token, JWT_SECRET, (err, user) => {
    if (err) {
      return res.status(403).json({ error: "Token invalide" });
    }

    const userId = user.userId;

    if (!isDbConnected) {
      return res.json({
        id: userId,
        email: user.email,
        name: user.name || "Utilisateur",
        phone: "+212 6 XX XX XX XX"
      });
    }

    db.query(
      "SELECT id, email, name, phone FROM users WHERE id = ?",
      [userId],
      (err, results) => {
        if (err || results.length === 0) {
          return res.status(404).json({ 
            success: false, 
            message: "Utilisateur non trouvé" 
          });
        }
        res.json(results[0]);
      }
    );
  });
});
